    def black_scholes(self, S, K, T, r, sigma, option_type='call'):
        """Calculate Black-Scholes option price and delta.
        
        Args:
            S: Current stock price
            K: Strike price
            T: Time to expiration in years
            r: Risk-free interest rate
            sigma: Volatility
            option_type: 'call' or 'put'
        
        Returns:
            tuple: (price, delta)
        """
        if T <= 0:
            # Handle expiration
            if option_type == 'call':
                return max(0, S - K), 1.0 if S > K else 0.0
            else:
                return max(0, K - S), -1.0 if S < K else 0.0
                
        d1 = (np.log(S/K) + (r + sigma**2/2)*T) / (sigma*np.sqrt(T))
        d2 = d1 - sigma*np.sqrt(T)
        
        if option_type == 'call':
            price = S*norm.cdf(d1) - K*np.exp(-r*T)*norm.cdf(d2)
            delta = norm.cdf(d1)
        else:
            price = K*np.exp(-r*T)*norm.cdf(-d2) - S*norm.cdf(-d1)
            delta = -norm.cdf(-d1)
            
        return price, delta

    def test_single_leg_margin(self):
        """Test margin calculation for a single leg option position."""
        # Test cases with different underlying prices and strikes
        test_cases = [
            # Short call scenarios
            (100.0, 90.0, OptionType.CALL, PositionSide.SHORT),  # ITM short call
            (100.0, 100.0, OptionType.CALL, PositionSide.SHORT), # ATM short call
            (100.0, 110.0, OptionType.CALL, PositionSide.SHORT), # OTM short call
        ]
        
        for underlying_price, strike, option_type, position_side in test_cases:
            with self.subTest(msg=f"{option_type} {position_side} - Strike: {strike}"):
                # Calculate option price using Black-Scholes
                entry_price, delta = self.black_scholes(
                    S=underlying_price,
                    K=strike,
                    T=30/365,  # 30 days
                    r=0.05,    # 5% risk-free rate
                    sigma=0.3, # 30% volatility
                    option_type='call' if option_type == OptionType.CALL else 'put'
                )
                
                print(f"\nTesting {option_type} {position_side}:")
                print(f"Entry Price: {entry_price:.4f}")
                print(f"Delta: {delta:.4f}")
                
                # Calculate margin using class method
                margin = SingleLegOptionPosition.calculate_margin(
                    quantity=1,
                    option_type=option_type,
                    position_side=position_side,
                    underlying_price=underlying_price,
                    entry_price=entry_price,
                    strike=strike
                )
                
                # Calculate expected margin
                if option_type == OptionType.PUT:
                    otm_amount = max(0, underlying_price - strike)
                else:  # CALL
                    otm_amount = max(0, strike - underlying_price)
                
                expected_margin = (
                    entry_price +  # Option price
                    max(
                        (0.15 * underlying_price - otm_amount),  # 15% of underlying minus OTM amount
                        (0.10 * underlying_price)  # 10% of underlying
                    )
                ) * 100  # Convert to dollars
                
                print(f"OTM Amount: {otm_amount:.2f}")
                print(f"Calculated Margin: {margin:.2f}")
                print(f"Expected Margin: {expected_margin:.2f}")
                
                self.assertEqual(margin, round(expected_margin, 2),
                    f"Margin mismatch for {option_type} {position_side} with strike {strike}")

    def test_single_leg_pnl(self):
        """Test P&L calculation for a single leg option position."""
        # Calculate entry and exit prices using Black-Scholes
        underlying_price = 100.0
        strike = 100.0
        entry_price, _ = self.black_scholes(
            S=underlying_price,
            K=strike,
            T=30/365,  # 30 days
            r=0.05,    # 5% risk-free rate
            sigma=0.3, # 30% volatility
            option_type='call'
        )
        
        # Calculate exit price with higher underlying
        exit_price, _ = self.black_scholes(
            S=underlying_price * 1.05,  # 5% increase in underlying
            K=strike,
            T=29/365,  # 29 days (1 day later)
            r=0.05,
            sigma=0.3,
            option_type='call'
        )
        
        # Test long call profit
        pnl = self.single_leg_long.calculate_pnl(exit_price=exit_price)
        expected_profit = (exit_price - entry_price) * 100 * self.single_leg_long.quantity
        self.assertEqual(pnl, expected_profit)

        # Test short put loss
        entry_price, _ = self.black_scholes(
            S=underlying_price,
            K=strike,
            T=30/365,
            r=0.05,
            sigma=0.3,
            option_type='put'
        )
        
        # Calculate exit price with lower underlying
        exit_price, _ = self.black_scholes(
            S=underlying_price * 0.95,  # 5% decrease in underlying
            K=strike,
            T=29/365,
            r=0.05,
            sigma=0.3,
            option_type='put'
        )
        
        pnl = self.single_leg_short.calculate_pnl(exit_price=exit_price)
        expected_loss = (entry_price - exit_price) * 100 * self.single_leg_short.quantity
        self.assertEqual(pnl, expected_loss)

            (5.0, 6.0, 2, PositionSide.LONG, OptionType.CALL, 196.44),  # 2 contracts
            (5.0, 4.0, 3, PositionSide.SHORT, OptionType.PUT, 294.66),  # 3 contracts
        ]

        for entry_price, exit_price, quantity, position_side, option_type, expected_pnl in test_cases:
            # Create test position
            position = SingleLegOptionPosition(
                trade_id=1,
                option_strategy=OptionStrategy.LONG_CALL if position_side == PositionSide.LONG else OptionStrategy.SHORT_CALL,
                quantity=quantity,
                option_type=option_type,
                position_side=position_side,
                entry_date=pd.Timestamp('2023-01-01'),

                entry_delta=0.5,
                entry_dte=30,
                underlying_entry=95.0
            )

            fees = 1.78 * quantity

    def test_buying_power_updates(self):
        """Test that buying power is correctly updated during position closure."""
        initial_bp = 10000
        
        # Test long position closure (should add exit premium to BP)
        long_pos = SingleLegOptionPosition(
            option_strategy=OptionStrategy.LONG_CALL,
            trade_id=1,
            quantity=1,
            option_type=OptionType.CALL,
            position_side=PositionSide.LONG,
            entry_date=pd.Timestamp('2023-01-01'),
            entry_price=5.0,
            strike=100.0,
            expire_date=pd.Timestamp('2023-01-31'),
            entry_delta=0.5,
            entry_dte=30,
            underlying_entry=95.0
        )
        
        result = long_pos.close(
            option_chain=self.data['option_chain'],
            underlying_price_history=self.data['underlying_price_history'],
            option_bp=initial_bp
        )
        self.assertIsNotNone(result)
        trade_result, transaction = result
        # BP should increase by exit premium and decrease by fees
        self.assertGreater(transaction['bp'], initial_bp)

        # Test short position closure (should subtract exit premium from BP and return margin)
        short_pos = SingleLegOptionPosition(
            option_strategy=OptionStrategy.SHORT_CALL,
            trade_id=2,
            quantity=1,
            option_type=OptionType.CALL,
            position_side=PositionSide.SHORT,
            entry_date=pd.Timestamp('2023-01-01'),
            entry_price=5.0,
            strike=100.0,
            expire_date=pd.Timestamp('2023-01-31'),
            entry_delta=0.5,
            entry_dte=30,
            underlying_entry=95.0
        )
        
        initial_bp = 10000
        result = short_pos.close(
            option_chain=self.data['option_chain'],
            underlying_price_history=self.data['underlying_price_history'],
            option_bp=initial_bp
        )
        self.assertIsNotNone(result)
        trade_result, transaction = result
        # BP should decrease by exit premium, increase by margin return, and decrease by fees
        final_bp = transaction['bp']
        self.assertNotEqual(final_bp, initial_bp)  # BP should change
        self.assertTrue(
            final_bp > initial_bp - (abs(transaction['exit_price']) * 100) + transaction['capital_used'] - trade_result['fees']
        )

    def test_margin_required_property(self):
        """Test the margin_required property calculation."""
        # Test long call - should have no margin requirement
        long_call = SingleLegOptionPosition(
            option_strategy=OptionStrategy.LONG_CALL,
            trade_id=1,
            quantity=1,
            option_type=OptionType.CALL,
            position_side=PositionSide.LONG,
            entry_date=pd.Timestamp('2023-01-01'),
            entry_price=5.0,
            strike=100.0,
            expire_date=pd.Timestamp('2023-01-31'),
            entry_delta=0.5,
            entry_dte=30,
            underlying_entry=100.0
        )
        self.assertEqual(long_call.margin_required, 0)

        # Test short put - should use IB's formula
        short_put = SingleLegOptionPosition(
            option_strategy=OptionStrategy.SHORT_PUT,
            trade_id=2,
            quantity=1,
            option_type=OptionType.PUT,
            position_side=PositionSide.SHORT,
            entry_date=pd.Timestamp('2023-01-01'),
            entry_price=5.0,
            strike=100.0,
            expire_date=pd.Timestamp('2023-01-31'),
            entry_delta=-0.5,
            entry_dte=30,
            underlying_entry=100.0
        )
        
        # Calculate expected margin using IB's formula
        otm_amount = max(0, short_put.underlying_entry - short_put.strike)
        expected_margin = (
            short_put.entry_price +
            max(
                (0.15 * short_put.underlying_entry - otm_amount),
                (0.10 * short_put.underlying_entry)
            )
        ) * 100 * short_put.quantity
        
        self.assertEqual(short_put.margin_required, round(expected_margin, 2))

        # Test short call - should use IB's formula
        short_call = SingleLegOptionPosition(
            option_strategy=OptionStrategy.SHORT_CALL,
            trade_id=3,
            quantity=1,
            option_type=OptionType.CALL,
            position_side=PositionSide.SHORT,
            entry_date=pd.Timestamp('2023-01-01'),
            entry_price=5.0,
            strike=100.0,
            expire_date=pd.Timestamp('2023-01-31'),
            entry_delta=0.5,
            entry_dte=30,
            underlying_entry=100.0
        )
        
        # Calculate expected margin using IB's formula
        otm_amount = max(0, short_call.strike - short_call.underlying_entry)
        expected_margin = (
            short_call.entry_price +
            max(
                (0.15 * short_call.underlying_entry - otm_amount),
                (0.10 * short_call.underlying_entry)
            )
        ) * 100 * short_call.quantity
        
        self.assertEqual(short_call.margin_required, round(expected_margin, 2))

        # Test margin for different moneyness levels
        test_cases = [
            # (underlying, strike, option_type, expected_otm)
            (100.0, 90.0, OptionType.PUT, 10.0),   # OTM put
            (100.0, 110.0, OptionType.PUT, 0.0),   # ITM put
            (100.0, 90.0, OptionType.CALL, 0.0),   # ITM call
            (100.0, 110.0, OptionType.CALL, 10.0), # OTM call
        ]

        for underlying, strike, option_type, expected_otm in test_cases:
            position = SingleLegOptionPosition(
                option_strategy=OptionStrategy.SHORT_CALL if option_type == OptionType.CALL else OptionStrategy.SHORT_PUT,
                trade_id=4,
                quantity=1,
                option_type=option_type,
                position_side=PositionSide.SHORT,
                entry_date=pd.Timestamp('2023-01-01'),
                entry_price=5.0,
                strike=strike,
                expire_date=pd.Timestamp('2023-01-31'),
                entry_delta=0.5 if option_type == OptionType.CALL else -0.5,
                entry_dte=30,
                underlying_entry=underlying
            )

            # Calculate expected margin
            expected_margin = (
                position.entry_price +
                max(
                    (0.15 * underlying - expected_otm),
                    (0.10 * underlying)
                )
            ) * 100 * position.quantity

            self.assertEqual(position.margin_required, round(expected_margin, 2),
                f"Margin mismatch for {option_type} with strike {strike} and underlying {underlying}")
